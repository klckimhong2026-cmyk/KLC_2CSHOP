            document.getElementById('adminLoginModal').classList.add('hidden');
            document.getElementById('adminPasswordInput').value = '';
        }

        function handleAdminLogin(event) {
            event.preventDefault();
            const pass = document.getElementById('adminPasswordInput').value;
            if (pass === '12345') {
                closeAdminLogin();
                openAdminPanel();
            } else {
                alert('លេខសម្ងាត់មិនត្រឹមត្រូវ! (សូមប្រើ: 12345)');
            }
        }

        function openAdminPanel() {
            document.getElementById('adminCurrentQrPreview').src = storeData.payment.qrImage;
            document.getElementById('adminMerchantNameInput').value = storeData.payment.merchantName;
            document.getElementById('adminAccountKHRInput').value = storeData.payment.accountKHR;
            document.getElementById('adminAccountUSDInput').value = storeData.payment.accountUSD;

            const gameSelector = document.getElementById('adminGameSelector');
            gameSelector.innerHTML = '';
            storeData.games.forEach(g => {
                const opt = document.createElement('option');
                opt.value = g.id;
                opt.textContent = g.name;
                gameSelector.appendChild(opt);
            });

            loadAdminPackagesForGame();
            document.getElementById('adminPanelModal').classList.remove('hidden');
        }

        function closeAdminPanel() {
            document.getElementById('adminPanelModal').classList.add('hidden');
        }

        let tempNewQrBase64 = null;
        function previewNewQrUpload(event) {
            const file = event.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    tempNewQrBase64 = e.target.result;
                    document.getElementById('adminCurrentQrPreview').src = tempNewQrBase64;
                };
                reader.readAsDataURL(file);
            }
        }

        function savePaymentSettings() {
            if (tempNewQrBase64) {
                storeData.payment.qrImage = tempNewQrBase64;
            }
            storeData.payment.merchantName = document.getElementById('adminMerchantNameInput').value;
            storeData.payment.accountKHR = document.getElementById('adminAccountKHRInput').value;
            storeData.payment.accountUSD = document.getElementById('adminAccountUSDInput').value;

            alert('បានរក្សាទុកការផ្លាស់ប្តូរ QR Code និងព័ត៌មានទូទាត់ដោយជោគជ័យ!');
        }

        function loadAdminPackagesForGame() {
            const gameId = document.getElementById('adminGameSelector').value;
            const game = storeData.games.find(g => g.id === gameId);
            const tbody = document.getElementById('adminPackagesTableBody');
            tbody.innerHTML = '';

            if (!game) return;

            game.packages.forEach((pkg, index) => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td class="p-3"><input type="text" value="${pkg.name}" id="pkgName_${gameId}_${index}" class="bg-slate-950 border border-slate-800 rounded px-2 py-1 text-white w-full"></td>
                    <td class="p-3"><input type="number" step="0.01" value="${pkg.usd}" id="pkgUsd_${gameId}_${index}" class="bg-slate-950 border border-slate-800 rounded px-2 py-1 text-cyan-400 w-24"></td>
                    <td class="p-3"><input type="text" value="${pkg.khr}" id="pkgKhr_${gameId}_${index}" class="bg-slate-950 border border-slate-800 rounded px-2 py-1 text-slate-300 w-32"></td>
                    <td class="p-3 text-center">
                        <button onclick="updatePackageData('${gameId}', ${index})" class="bg-blue-600 hover:bg-blue-500 text-white px-3 py-1 rounded text-xs font-bold">កែប្រែ</button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function updatePackageData(gameId, index) {
            const game = storeData.games.find(g => g.id === gameId);
            if (!game) return;

            const newName = document.getElementById(`pkgName_${gameId}_${index}`).value;
            const newUsd = parseFloat(document.getElementById(`pkgUsd_${gameId}_${index}`).value);
            const newKhr = document.getElementById(`pkgKhr_${gameId}_${index}`).value;

            game.packages[index].name = newName;
            game.packages[index].usd = newUsd;
            game.packages[index].khr = newKhr;

            alert(`បានធ្វើបច្ចុប្បន្នភាពកញ្ចប់ "${newName}" រួចរាល់!`);
        }
